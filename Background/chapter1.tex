Multicategories are a generalization of categories. In 
categories, arrows have a single source/domain and a single target/codomain. In multicategories, the arrows have a finite sequence of sources/inputs and a single target/output. 
\begin{defn}
    A \emph{multicategory} $\M$ consist of a set $S$ of objects and for each $n\geq 0$ and each sequence $s_1,\ldots, s_n,t$ of objects a set
    \[\M(s_1,\ldots,s_n;t) \] 
    of multiarrows/operations, thought of as taking $n$ inputs objects $s_1,\ldots,s_n$ respectively to the output object $t$. Moreover, there are three kinds of structure maps:
    \begin{itemize}
        \item For each object $s$, an identity/unit $1_s\in \M(s;s)$
        \item For any sequence of objects $s_1,\ldots,s_n,t$ and any $n$-tuple of sequences $d^i_1,\ldots,d^i_{k_i}$ of objects for $i=1,\ldots,n$, a composition function of multiarrows/operations 
        \[
        \gamma:\M(s_1,\ldots,s_n;t)\times\prod_{i=1}^{n}\M(d^i_1,\ldots,d^i_{k_i};s_i)\rightarrow\M(d^1_1,\ldots,d^1_{k_1},d^2_1,\ldots,d^n_{k_n};t),
        \]
        usually written $p\circ(q_1,\ldots,q_n):=\gamma(p,q_1,\ldots,q_n)$ or even just $p(q_1,\ldots,q_n)$
    \end{itemize}
    Furthermore these structure maps have to satisfy a number of axioms, which express that composition is unital and associative \citep{HM}
    % {\bf ADD SPECIFIC CITATION WHERE THE FULL DEFINITION APPEARS.}
\end{defn}
\begin{defn}
    A (colored) operad $\boldsymbol{P}$ is a multicategory with a symmetric action on the set of multiarrows. That is:
    \begin{itemize}
        \item For each permutation $\sigma\in \Sigma_n$, a map
        \[
        \sigma^* : \boldsymbol{P}(\lst{c}{n};c)\rightarrow \boldsymbol{P}(c_{\sigma(1)},\ldots,c_{\sigma(n)};c)
        \]
        usually written $\sigma^*p=p\circ \sigma$
    \end{itemize}
    such that these maps are compatible with the composition and the units giving a right action of the symmetric groups on the sets of multiarrows.
\end{defn}

We will be calling multicategories when we are not interested in the symmetric action, and operads when we are.
\begin{rmk}
Let $\M$ be a multicategory. Given multiarrows $f\in\M(s_1,\ldots,s_n;t)$ and $g\in\M(d_1,\ldots,d_l;s_i)$ for some $1 \leq i \leq n$. It will be convenient to have the following notation:
\[
\iof{f}{g} := f(1_{s_1},\ldots,g,\ldots,1_{s_n})
\]
Thus, $\iof{f}{g}$ is the composition of $f$ with $g$ in the $i$th variable.
\end{rmk}

\begin{ex}
    You can think of $n$-ary multiarrows in $\M(s_1,\ldots, s_n;t)$ as n-ary operations. Let $M$ be the multicategory with a single object $X$. Then the $n$-ary multiarrows in $M(X,\dots,X;X)$define to be the set of functions 
    \[
    f:X\times \cdots \times X \rightarrow X 
    \]
    where the domain is the n-fold cartesian product of $X$ with itself.
\end{ex}
\begin{ex} 
    We can up the level from the previous example, by fixing a family of sets $ \{X_c\}_{c\in C}$ indexed by a set of objects $C$. Then you can think of the multiarrows in $\M(c_1,\ldots,c_n;c)$ as n-ary operations, taking inputs of types $\lst{c}{n}$ respectively to an output of type $c$.

    For example, for a fixed family of sets $\{X_c\}_{c\in C}$, define a multicategory $\M$ whose objects is the set of colors $C$,  and whose multiarrows are defined by declaring $\M(c_1,\ldots, c_n;c)$ to be the set of functions 
     \[
     p:X_{c_1}\times \cdots \times X_{c_n}\rightarrow X_c
    \]
    Composition in $\M$ is defined as composition of functions.
\end{ex}

We can observe that multicategories form a category $\MCat$.


\begin{defn}
For two multicategories $\M$ and $\mathcal{N}$, a \emph{multifunctor} $\phi:\M\rightarrow\mathcal{N}$ consist of:
\begin{itemize}
    \item A function $f:O_{\M}\rightarrow O_{\mathcal{N}}$ between the corresponding sets of objects
    \item For each sequence $s_1,\ldots,s_n,t$ of objects of $\M$, a map 
    \[
        \phi_{\lst{s}{n},t}:\M(\lst{s}{n};t)\rightarrow \mathcal{N}(\flst{s}{n}{f};f(t)).
    \]
\end{itemize}
So that these maps are compatible with compositions and units in the obvious way.
\end{defn}
\subsection{The Tensor Product of Operads}
\label{sec:The Tensor Product of Operads}

Let $M$ and $N$ be operads. Their Boardman-Vogt tensor product $M \bv N$ is a operad defined to make sense of what a $M$-algebra in the category of $N$-algebras is and vice versa. We will not talk about the algebras of a multicategories here, but we refer the reader to [look for reference] % citep{?} and \citep{?} %} for more details.
\begin{defn}
    We define the \emph{Boardman-Vogt tensor} $M \bv N$ in terms of generators and relations. The set of objects of this multicategory is the product of the sets of objects of $M$ and $N$; if $x$ and $y$ are objects of $M$ and $N$ respectively, we write $x\otimes y$ for the corresponding object of $M \bv N$. The generating multiarrows of $M \bv N$ are of two types:
\begin{itemize}
    \item For each multiarrow $f \in M (\lst{x}{n},x)$ and object $y$ of $N$, a multiarrow \[
    f \otimes y \in (M\bv N)(\tflst{x}{n}{y};x\otimes y)
    \]
    \item For each object $x$ of $M$ and multiarrow $g \in N(\lst{y}{m};y)$, a multiarrow
    \[
    x \otimes g \in (M\bv N)(\lst{x\otimes y}{m}; x\otimes y)
    \]
\end{itemize}
The relations to be satisfied by these generators are the following:
\begin{enumerate}[label = (\arabic*)]
    \item $(f\otimes y)(\tflst{f}{n}{y}) = f(\lst{f}{n})\otimes y$
    \item $(x\otimes g)(\lst{x\otimes g}{m})= x \otimes g(\lst{g}{m})$
    \item $\sigma^*(f \otimes y) = (\sigma^* f)\otimes y$ for $\sigma \in \Sigma_n$
    \item $\sigma^*(x \otimes g) = x \otimes (\sigma^* g)$ for $\sigma \in \Sigma_m$
    \item $(f \otimes y)(\tflst{x}{n}{g}) = \sigma^*_{n,m}((x \otimes g)(\lst{f\otimes y}{m}))$
\end{enumerate}
\end{defn}

Note that relations (1) and (2) say precisely the condition that for any object $y$ of $N$ there is a multifunction $-\otimes y:M \rightarrow M \bv N$. Similarly, for any object $x$ of $M$ there is a multifunction $x \otimes -:N \rightarrow M \bv N$, this time given by relations (2) and (4). Relation (5) is the most interesting; most often referred to it as the \emph{Boardman-Vogt interchange relation}. The permutation $\sigma_{n,m} \in \Sigma_{nm}$ is the one that makes sense of formula (5), i.e. the element relating the sequences $(\lst{x_1 \otimes y}{m},\dots,\lst{x_n \otimes y}{m})$ and $(\tflst{x}{n}{d_1},\dots,\tflst{x}{n}{d_m})$.

\section{Trees}
\label{sec:trees}
We will do a quick detour to talk about our main combinatorial object that will help us with studying multicategories: trees. We are mainly interested in finite rooted trees. So we don't get confused with the more general notion of tree in graph theory, we will give a precise definition of what we mean by a tree in this thesis. We will see that these trees give rise to free multicategories whose objects are the edges of the tree and morphisms are generated by the vertices of the tree.
\begin{defn}
    A \emph{tree} $T$ consist of a finite set $V$ of \emph{vertices}, a nonempty finite set $E$ of \emph{edges}, a distinguished element $r\in E$ called the \emph{root}, together with the following data:
    \begin{enumerate}[label= (\arabic*)]
        \item A function $I:E-\{r\}\rightarrow V$, which we think of as assigning to an edge $e$ the vertex $I(e)$ of which is an input. The way to think about this function is that "$I(e)=f$ if and only if $e$ is \emph{an} input edge of $f$"
        \item An injective function $O:V\rightarrow E$, assigning to each vertex $g$ its output edge $O(g)$. The way to think about this function is that "$O(f)=x$ if and only if $x$ is \emph{the} output edge of $f$"
    \end{enumerate}
For each edge $e$ other than the root $r$, we obtain a sequence of edges starting at $e$ by repeatedly applying $O\circ I$. We demand that this sequence ends in the root $r$ after finitely many steps, for an arbitrary starting edge.
\end{defn}

As we can see the main difference between our definition of a tree and the usual definition of a tree in graph theory is that we allow for "orphan"/"leaf" edges that are not attached to any vertex as an output edge, and a unique designed edge that will not be attached to any vertex as an input edge (the root). This special edge also gives us a natural orientation of the tree, where we think of the root as the bottom and the leaves as the top, and thus a partial ordering on both the set of edges and the set of vertices.

Now I will introduce some terminology to talk about the different parts of a tree. The edges in the complement of the image of $O$ are called \emph{leaves} of the tree $T$. The vertices in the complement of the image of $I$ are called \emph{stumps}, or \emph{nullary vertices.} An \emph{outer edge} is an edge that is either the root or a leaf. An \emph{inner edge} is any other edge of $T$. Such an edge is both an output edge and an input edge for some other vertex. The vertices come in two kinds, \emph{external vertices} which only have one inner edge attached to them, and \emph{internal vertices} 
When writing $T$ for a tree, we will usually write $E(T)$ and $V(T)$ for its sets of edges and vertices, respectively.

\begin{defn}
    An \emph{embedding} $\psi:S\rightarrow T$ of trees consists of injective maps $\psi_V:V(S)\rightarrow V(T)$ and $\psi_E:E(S)\rightarrow E(T)$ such that for any vertex $f$ of $S$, the map $\psi_E$ gives a bijection between $O^{-1}(f)$ and $O_{-1}(\psi_V(f))$, and $\psi_E(O(f))=O(\psi_V(f))$. If an embedding is bijective on both edges and vertices, then it is an isomorphism of trees.
\end{defn} 

If an embedding of trees $\iota:S\rightarrow T$ is given by inclusion maps of $E(S)\subseteq E(T)$ and $V(S)\subseteq V(T)$, we will say that $S$ is a \emph{subtree} of $T$. Then we can see that any embedding $\psi:S\rightarrow T$ factors as an isomorphism followed by the inclusion of a subtree.
\begin{defn}
A tree $T$ defines an operad $\dend(T)$ as follows: 
\begin{itemize}
    \item The objects of $\dend(T)$ are the edges of $T$.
    \item $\dend(T)(e_1,\ldots,e_n;e)$ is either empty or a singleton if there exist a subtree $C_n$ with leaves $\lst{e}{n}$ and root $e$. This set is generated by the vertices of the tree and freely composing them to get the other subtrees' operations.  
    \item Composition is given by the tree structure in the obvious way. Expanding the details of it, we get the composition operation/multiarrow by identifying the root of the precomposing operation with a leave of the postcomposing operation and squashing this new inner edge and identifying the two vertices to the composition of the two corresponding operations/multiarrows.
    \item The action of $\Sigma_n$ on n-ary operation is given by the relabeling of the input edges of a vertex that defines a generating operation. The other subtrees' operation follow, making sure we respect composition ordering of inputs.
\end{itemize}
\end{defn}

\section{Category of trees $\dend$}
\label{sec:dset}

Dendroidal sets are a generalization of simplicial sets that are particularly well suited to study operads/multicategories and multicategories. Just as simplicial sets are presheaves on the simplex category $\simp$, dendroidal sets are presheaves on the dendroidal category $\dend$.
\begin{defn}
    The category $\dend$ consist of:
    \begin{enumerate}
        \item Objects: finite rooted trees
        \item Morphisms: for two trees $S$ and $T$, a morphism from $S$ to $T$ is a multifunctor $\dend(S)\rightarrow \dend(T)$ between the corresponding Operads.
    \end{enumerate}
    Another way to say this is that $\dend$ is the full subcategory of $\cat{Op}$, the category of operads, spanned by the multicategories of the form $\dend(T)$ for some tree $T$.
\end{defn}
 
The linear trees $\left[n\right]$ define a full subcategory of $\dend$ isomorphic to the simplex category $\simp$. Thus, there is an inclusion functor $i : \simp \rightarrow \dend$. This inclusion is compatible with the inclusion of $\cat{Cat}$ into $\MCat$ in the sense that the following diagram commutes:

\section{Faces and Degeneracies in $\dend$}
\label{sec:faces and degeneracies}
Recall that, the simplex category $\simp$ has two important families of morphisms, the face maps and the degeneracy maps. These maps generate all other morphisms in $\simp$. The good news for $\dend$ being some multicategorical extension of $\simp$ is that it also has two families of morphisms that generate all other morphisms in $\dend$. Like for the simplex category, the face maps in $\Omega$ have two kinds: 
\begin{itemize}
    \item The outer face maps
    \item The inner face maps
\end{itemize}
The other calls of maps are again called degeneracies.
\begin{defn}[Outer Faces]
For a tree $T$ and an external vertex $f$ of T, write $\partial_f T$ for the tree obtained by pruning away $f$ and all the outer edges attached to it. This vertex could be any leaf vertex or it can be the vertex attached to the root of $T$, provided it is only attached to one inner vertex of $T$. We write
\[
\partial_f T \xrightarrow{\delta_f} T
\] 
for the inclusion of this tree and refer to it as an \emph{outer face}. If $f$ is a leaf vertex we will sometimes specifically speak of a \emph{leaf face}, and similarly for a \emph{root} face. 
\end{defn}

     Corollas, by our definition, are trees with only one unique vertex and $n$ leaves. This vertex in a corolla is not an external vertex because it is not attached to any interior edges. Nevertheless by convention, $C_n $ has $n+1$ different maps $\mu \rightarrow C_n$ corresponding to the inclusion of each of the $n$ leaves and the root. We define all of these to be outer faces.
\end{rmk}

\begin{defn}[Inner Faces]
   For a tree $T$ and an inner edge $x$, let $\partial_x T$ denote the face of $T$ obtained by contracting the edge $x$, and identifying the vertices $f$ and $g$ at either end of $x$. This is an \emph{inner face} of $T$. The associated inner face map denoted
    \[
        \partial_xT\xrightarrow{\delta_x} T
    \]
    Should be understood not in terms of the combinatorics of trees but in terms of their associated operads. In this framework, we see that we send the new vertex in $\partial_xT$ to the composition in $\dend(T)$ of the operations correspondingly given by $f$ and $g$ along $x$. Thus the image of the new vertex in $\partial_x T $ can be identified with the subtree of $T $ generated by the edge x. A morphism of these kind will be referred to as an \emph{inner face}.
 
\end{defn}

\section{Dendroidal Sets}
\label{Dendroidal set}
As no surprise to the reader, the definition of a \emph{dendroidal set} is analogous to the definition of a simplicial set.
\begin{defn}
    A dendroidal set is a functor 
    \[X\ :\ \dend^{\op} \xrightarrow{} \Set  \]
    Dendroidal sets form the objects of a category $\dset$ whose morphisms are the natural transformations between them.
\end{defn}

Expounding out the definition of a dendroidal sets, we can see a dendroidal set is given by a family of sets $X_T$, one for each $T\in \dend$, together with a map 
\[\alpha^*\ : \ X_T \rightarrow X_S \]
for every morphism $\alpha : S \rightarrow T$ in $\dend$. These maps should be functorial. The elements of $X_T$ are called \emph{dendrices} of $X$ of shape $T$, or simply $T$-dendrices of $X$.

In particular, each set $X_T$ carries an action of the group $\Aut(T )$ and the dendroidal set $X$ is completely determined by the collection of all these $\Aut(T)$-sets together with the elementary face and degeneracy operators of the three kinds we saw before, satisfying the \emph{dendroidal identities}. dual to the identities previously mentioned.

\begin{ex}
    Any tree $T$ of $\dend$ gives rise to a representable presheaf which we denote by $\dend[T]$. As in any presheaf category this are defined by
    \[
    \dend [T]_S = \Hom_{\dend}(S,T) 
    \]
\end{ex}

\begin{ex}
    One can form limits and colimits of diagrams of dendroidal sets by calculating them `pointwise', as in any presheaf category. For instance, we can form the product $X\times Y $ of dendroidal sets $X$ and $Y $ as 
    \[(X \times Y )_T = X_T \times Y_T  \]   
\end{ex}

\begin{ex}
    In particular, each face $\delta_x : \partial_x T \rightarrow   T   $, where $x$ is an inner edge or outer vertex, defines a monomorphism of dendroidal sets
    \[\dend[\partial_x T] \rightarrow \dend[T ] .\]
    We will write $\partial_x\dend[T]\subseteq\dend[T]$ for the image of this morphism and call it the \emph{face of $T$ opposite $x$}. The union of all these (inner and outer faces) defines the \emph{boundary} $\partial\dend[T]$ of $\dend[T]$. Note that these faces and the boundary can be explicitly described as follows:
    \[
        \begin{aligned}
            (\partial_x\dend[T])_S &= \{\alpha:S\rightarrow T\ |\ \text{the edge } x \text{ is not contained in the image of } \alpha\},\\
            (\partial_f\dend[T])_S &=\{\alpha:S\rightarrow T\ |\ \text{no subtree } \alpha(g) \text{ contains the vertex } f\},\\
            (\partial\dend[T])_S &=\{\alpha : S \rightarrow T\ |\ \alpha \text{ factors through a proper inclusion } R \rightarrow T \}.
        \end{aligned}
    \]

\end{ex}

Following suit with simplicial sets, we can define the notion of degenerate and non-degenerate dendrices. We will use them later to give a more explicit description of the left adjoint $h:\dset\rightarrow\cat{Op}$ to the nerve functor $\textup{N}:\cat{Op}\rightarrow\dset$.
\begin{defn}
    Let $D$ be a dendroidal set. A dendrex $d\in D_T$ is called \emph{degenerate} if there is a degeneracy map $\sigma: T \rightarrow S$ which is not an isomorphism and a dendrex $y\in X_S$ with $\sigma^*y=d$. A dendrex is \emph{non-degenerate} if such a pair does not exist. 
\end{defn}

\section{Tensor Product in $\dset$}
Recall from \ref{sec:The Tensor Product of Operads} that the category of operads $\cat{Op}$ has a monoidal structure given by the Boardman-Vogt tensor product. Since the nerve functor $\textup{N}:\cat{Op}\rightarrow\dset$ is fully faithful, we can transport this monoidal structure to $\dset$ and obtain a monoidal structure on it. We will do this by by first define a functor \[
ten: \dend \times \dend \rightarrow \dset
\]
by assigning to a pair of tree $(S,T)$ the corresponding free operads $\freeOp S$ and $\freeOp T$, forming their tensor product $\freeOp S\bv\freeOp T$ and finally taking the dendroidal nerve of the resulting operad: 
\[
    ten(S,T)=\textup{N}(\freeOp S\bv\freeOp T).     
\]
By the method of Kan extension, this functor determines a functor 
\[
\otimes:\dset \times \dset \rightarrow \dset
\]
uniquely determined up to natural isomorphism by the requirement that it preserves colimits in each variable and that it coincides with $ten$ on representables, i.e.
\begin{align*}
    \freeOp{S}\otimes\freeOp{T} &= \textup{N}(\freeOp S\bv\freeOp T),\\
    \varinjlim_i (X \otimes Y_i) &\simeq X \otimes \varinjlim_i Y_i,\\
    \varinjlim_i (X_i \otimes Y) &\simeq \varinjlim_i X_i \otimes Y.
\end{align*}
The category $\dset$, being a presheaf category, is Cartesian closed and therefore has inner hom objects adjoint to the cartesian product. However, this inner homs do not model the correct type of morphisms that we want for the category of dendroidal sets to be a combinatorial model for the homotopy theory of operads. Instead, we need an inner hom object that adjoint to the tensor product we have just defined.
\begin{defn}
The inner hom object which is right adjoint to the tensor product is denoted by $\dHom$. This internal hom is characterized by natural isomorphism 
\[
\dset(X\otimes Y, Z) \simeq \dset(X,\dHom(Y,Z))
\]
for dendroidal sets $X,Y,Z$.
\end{defn}

We will use extensively this object and the underlying simplicial set of these hom object.
\[
\dhom(Y,Z):= i^*\dHom(Y,Z).
\]    

\begin{prop}\label{prop tensor in dset}
    The tensor product of dendroidal sets satisfies the following  properties:
    \begin{enumerate}[label=(\roman*)]
        \item It is symmetric 
        \item The functor $h:\dset \rightarrow \cat{Op}$ commutes with tensor products, i.e. there is a natural isomorphism of operads $h(X\otimes Y)\simeq h(X)\bv h(Y)$.        
        \item The functor $i_!:\sset \rightarrow \dset$ send product to tensor products, i.e. there is a natural isomorphism $i_!(M \times N) \simeq i_!(X) \otimes i_!(Y)$ for simplicial sets $M,N$.
    \end{enumerate}
\end{prop}
\begin{proof}
    Since the functors $\otimes,\ h, $ and $i_!$ are all compatible with colimits it suffices to establish these natural isomorphism for representables. Then $(i)$ follows from the symmetry of the Boardman-Vogt tensor product. For $(ii)$, recall from Remark \ref{DendroidalSetsandOperads} that the nerve functor $\textup{N}$ is fully faithful, so that the counit $\epsilon:h\Nrv{}\rightarrow \textup{id}$ is a natural isomorphism. It follows that
    \begin{align*}
    h(\drep{S}\otimes \drep{T})&=h\Nrv{}(\freeOp{S}\bv \freeOp{T})\\
    &\simeq \freeOp{S}\bv \freeOp{T}\\
    h(\drep{S}\otimes \drep{T})&=h (\Nrv{}\freeOp S)\bv h (\Nrv{}\freeOp T)    
    \end{align*}
    Finally, $(iii)$ follows from the fact that for categories $\cat{C}$ and $\cat{D}$ the tensor product $\cat C \bv \cat D $ coincides with the cartesian product $\cat{C} \times \cat{D}$, so that for simplicial sets $M,N$ we have 
    \begin{align*}
    i_!(M \times N)_{[n]} &= (M \times N)_n\\       
    &\simeq M_n \times N_n\\
    &\simeq i_!M_{[n]} \times i_!N_{[n]}\\
    &\simeq (i_!M \otimes i_!N)_{[n]}
    \end{align*}

\end{proof}

\begin{rmk}\label{rmk:notassociative}
    We will note that the tensor product of dendroidal sets is not associative up to coherent isomorphism. This is the main issue at hand we will need to address in the rest of the thesis. In good news, we will see that the tensor product is going to be associative up to homotopy, and that this is good enough for our purposes.
\end{rmk}
